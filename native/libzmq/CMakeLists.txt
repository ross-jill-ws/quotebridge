cmake_minimum_required(VERSION 3.20)
project(QuoteBridgeNative LANGUAGES C CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "This adapter requires MSVC Win32")
endif()
if(NOT ZMQ_SOURCE_DIR)
  message(FATAL_ERROR "Set ZMQ_SOURCE_DIR to the extracted upstream zeromq-4.3.5 folder")
endif()
set(BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(BUILD_STATIC ON CACHE BOOL "" FORCE)
foreach(option BUILD_TESTS WITH_PERF_TOOL ENABLE_CURVE WITH_LIBSODIUM WITH_TLS WITH_LIBBSD WITH_DOC WITH_DOCS ENABLE_DRAFTS ENABLE_WS ENABLE_CPACK ENABLE_CLANG ENABLE_PRECOMPILED)
  set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
add_subdirectory("${ZMQ_SOURCE_DIR}" upstream)
set_property(TARGET libzmq-static PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
add_library(quote_bridge_zmq SHARED adapter.cpp adapter.def)
target_link_libraries(quote_bridge_zmq PRIVATE libzmq-static)
target_compile_options(quote_bridge_zmq PRIVATE /W4 /WX /Oy-)
set_target_properties(quote_bridge_zmq PROPERTIES OUTPUT_NAME "libzmq-mt-4_3_5" MSVC_RUNTIME_LIBRARY MultiThreaded)
add_executable(native_probe probe.cpp)
target_compile_options(native_probe PRIVATE /W4 /WX /Oy-)
set_target_properties(native_probe PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreaded)
